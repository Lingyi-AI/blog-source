        static void wjtoshaizi(ref bool b,ref int zt,ref WanJia wanJia, ref WanJia diannao, Ditujiegoti ditujiegoti) 
        {
            if (wanJia.zt)
            {
                zt++;
                if (zt==2) 
                {
                    Console.ForegroundColor = ConsoleColor.Blue;
                    Console.SetCursorPosition(1, 25);
                    Console.Write("玩家暂停一回合");
                    zt = 0;
                    wanJia.zt = false;
                    b = !b;
                    return;
                }
            }
            else if (diannao.zt) 
            {
                zt++;
                if (zt == 2)
                {
                    Console.ForegroundColor = ConsoleColor.Magenta;
                    Console.SetCursorPosition(1, 25);
                    Console.Write("电脑暂停一回合");
                    zt = 0;
                    diannao.zt = false;
                    b = !b;
                    return;
                }
            }

            Random r = new Random();
            int wjsjs = 0;
            wjsjs =r.Next(1,7);

            if (b)
            {
                wanJia.x += wjsjs;
                b = !b;
            }
            else 
            {
                diannao.x += wjsjs;
                b = !b;
            }
            
            if (wanJia.x >= ditujiegoti.gezijiegotis.Length - 1) //判断到最后一格
            {
                wanJia.x = ditujiegoti.gezijiegotis.Length - 1;
                Console.SetCursorPosition(1, 28);
                Console.Write("游戏结束,玩家胜利");
            }
            else if (diannao.x >= ditujiegoti.gezijiegotis.Length - 1)
            {
                Console.SetCursorPosition(1, 28);
                Console.Write("游戏结束,电脑胜利");
            }

            Console.ForegroundColor = b ? ConsoleColor.Magenta : ConsoleColor.Blue;
            Console.SetCursorPosition(1, 25);
            Console.Write("{0}投出色子的点数为:{1}",b ? "电脑" : "玩家" ,wjsjs);

            Console.SetCursorPosition(1, 26);
            switch (ditujiegoti.gezijiegotis[wanJia.x].e_Gezi)//根据格子类型判断执行对应格子事件
            {
                case E_gezi.ptgezi:
                    Console.Write("{0}到达一个安全的位置", b ? "电脑" : "玩家");
                    break;
                case E_gezi.zantinzgezi:
                    if (b)
                    {
                        diannao.zt = true;
                    }
                    else
                    {
                        wanJia.zt = true;
                    }
                    Console.Write("{0}到达一个暂停的位置，暂停一回合", b ? "电脑" : "玩家");
                    break;
                case E_gezi.zhadangezi:
                    if (b)
                    {
                        diannao.x -= 5;
                        if (diannao.x <= 0)
                        {
                            diannao.x = 0;
                        }
                    }
                    else
                    {
                        wanJia.x -= 5;
                        if (wanJia.x <= 0)
                        {
                            wanJia.x = 0;
                        }
                    }
                    Console.Write("{0}到达一个炸弹的位置，倒退5格", b ? "电脑" : "玩家");
                    break;
                case E_gezi.shikogezi:
                    wjsjs = r.Next(1,100);

                    {

                        if (b)
                        {
                            diannao.x -= wjsjs;

                        else
                        {
                            wanJia.x -= wjsjs;
                        }
                        Console.Write("{0}到达一个时空隧道的位置,倒退{1}格", b ? "电脑" : "玩家", wjsjs);
                    }
                    else if (wjsjs >= 33 && wjsjs < 66)

                        if (b) 
                        {
                            diannao.zt=true;
                        }
                        else
                        {
                                wanJia.zt=true;

                        Console.Write("{0}到达一个时空隧道的位置,暂停一回合", b ? "电脑" : "玩家");

    struct Gezijiegoti//格子结构体
    {
        public E_gezi e_Gezi;
        public int x;
        public int y;

        public Gezijiegoti(int x, int y, E_gezi e_Gezi) //格子结构体构造函数
        {
            this.x = x;
            this.y = y;
            this.e_Gezi = e_Gezi;
        }
        public void huagezifangfa()
        {
            Console.SetCursorPosition(x, y);
            switch (e_Gezi)
            {
                case E_gezi.ptgezi:
                    Console.ForegroundColor = ConsoleColor.White;
                    Console.Write("□");
                    break;
                case E_gezi.zantinzgezi:
                    Console.ForegroundColor = ConsoleColor.Blue;
                    Console.Write("‖");
                    break;
                case E_gezi.zhadangezi:
                    Console.ForegroundColor = ConsoleColor.Red;
                    Console.Write("X");
                    break;
                case E_gezi.shikogezi:
                    Console.ForegroundColor = ConsoleColor.Yellow;
                    Console.Write("Y");
                    break;
            }
        }//画格子方法函数
    }
    struct Ditujiegoti//地图结构体
    {
        public Gezijiegoti[] gezijiegotis;

        public Ditujiegoti(int x, int y, int a)
        {
            gezijiegotis = new Gezijiegoti[a];
            Random r = new Random();
            int random;
            int posX = 0;
            int posY = 0;
            int buc = 2;
            for (int i = 0; i < a; i++) //格子是四种里的概率
            {
                random = r.Next(0, 101);
                if (random < 85 || i == 0 || i == gezijiegotis.Length - 1)//开始格和结束格不能为特殊格子
                {
                    gezijiegotis[i].e_Gezi = E_gezi.ptgezi;
                }
                else if (random >= 85 && random < 90)
                {
                    gezijiegotis[i].e_Gezi = E_gezi.zantinzgezi;
                }
                else if (random >= 90 && random < 95)
                {
                    gezijiegotis[i].e_Gezi = E_gezi.zhadangezi;
                }
                else { gezijiegotis[i].e_Gezi = E_gezi.shikogezi; }

                gezijiegotis[i].x = x;
                gezijiegotis[i].y = y;

                if (posX == 15)
                {
                    y++;
                    posY++;
                    if (posY == 2)
                    {
                        posX = 0;
                        posY = 0;
                        buc = -buc;
                    }
                }
                else
                {
                    x += buc;
                    posX++;
                }
            }

        }
        public void huaditufangda()//画地图方法
        {
            for (int i = 0; i < gezijiegotis.Length; i++)
            {
                gezijiegotis[i].huagezifangfa();
            }

        }
    }

            this.e_Wjdn = e_Wjdn;
            this.geziwz = geziwz;
            this.zt = zt;

            switch (e_Wjdn)
            {
                case E_wjdn.wanjia:
                    Console.SetCursorPosition(geziwz[x].x, geziwz[x].y);
                    Console.ForegroundColor = ConsoleColor.Cyan;

                    break;
                case E_wjdn.diannao:
                    Console.SetCursorPosition(geziwz[x].x, geziwz[x].y);
                    Console.ForegroundColor = ConsoleColor.Magenta;
                    Console.Write("B");
                    break;
            }
        }

    }
