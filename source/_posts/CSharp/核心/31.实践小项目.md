using System;

        static void Main(string[] args)
        {
            int Height = Console.BufferHeight;//缓冲区高,30
            int Width = Console.BufferWidth;//缓冲区宽,120
            Console.CursorVisible = false;//光标显示关闭

            //初始化
            E_changjingenum cjenum = E_changjingenum.kaishichangjing;
            E_erxyi erxyi = E_erxyi.xxiang1;
            ChengJingxzLei cjl = new ChengJingxzLei();
            DiTu ditu = new DiTu();
            TanChSheLei tcs = new TanChSheLei();
            ShiWuShuZuLei sw = new ShiWuShuZuLei();

            while (true) 
            {
                switch (cjenum)
                {
                    case E_changjingenum.kaishichangjing:
                        Console.Clear();
                        cjl.huizhi(ref cjenum,ref erxyi, Height, Width); //开始场景
                        break;
                    case E_changjingenum.youxichangjing:
                        Console.Clear();
                        ditu.huizhi(Height,Width);

                        while (true)//游戏内容主逻辑
                        {
                            DateTime sj = DateTime.Now;


                            tcs.panduanfangxiang();//判断方向方法
                            tcs.yido();//实现位置坐标移动方法
                            tcs.tcschsw(sw);
                            if (tcs.siwang()) //判断是否死亡
                            { 
                                cjenum = E_changjingenum.jieshuchangjing;
                                tcs.sheLeis = new SheLei[1]; //重新初始贪吃蛇
                                sw.swshuzu = null; //重新初始食物
                                tcs.sheLeis[0] = new SheLei(); //对贪吃蛇初始位置
                                tcs.sheLeis[0].wzxy.x = 50;
                                tcs.sheLeis[0].wzxy.y = 10;
                                tcs.sheLeis[0].tubiao = "G";
                                break; 
                            }
                            tcs.huizhi(ref tcs.yidocshu);//绘制方法
                            if (tcs.yidocshu > 15) //根据蛇移动的距离生成食物
                            { 
                                sw.ShengChengSW(tcs);
                                tcs.yidocshu = 0;
                            }
                            tcs.zhenkozhi(sj);//帧控制的方法
                        }

                        break;
                    case E_changjingenum.jieshuchangjing:
                        Console.Clear();
                        cjl.huizhi(ref cjenum, ref erxyi, Height, Width); //结束场景
                        break;
                }
            }
        }
    }

    {
        public void huizhi(ref E_changjingenum a,ref E_erxyi b,int h,int w)//实现开始游戏和结束游戏方法
        {
            Console.ForegroundColor = ConsoleColor.White;
            if (a == E_changjingenum.kaishichangjing)
            {
                Console.SetCursorPosition(w / 2, h / 2 - 7);
            }
            else { Console.SetCursorPosition(w / 2 - 1, h / 2 - 7); }
            Console.WriteLine(a == E_changjingenum.kaishichangjing ? "贪吃蛇" : "游戏结束");
            Console.ForegroundColor = b == E_erxyi.xxiang1 ? ConsoleColor.Red : ConsoleColor.White;
            if (a == E_changjingenum.kaishichangjing)
            {
                Console.SetCursorPosition(w / 2 - 1, h / 2 - 4);
            }
            else { Console.SetCursorPosition(w / 2 - 3, h / 2 - 4); }
            Console.WriteLine(a == E_changjingenum.kaishichangjing ? "开始游戏" : "回到开始界面");
            Console.ForegroundColor = b == E_erxyi.xxiang2 ? ConsoleColor.Red : ConsoleColor.White;
            Console.SetCursorPosition(w / 2 - 1, h / 2 - 2);
            Console.WriteLine("结束游戏");
            char c = Console.ReadKey(true).KeyChar;
            if (c == 'w')
            {
                b = E_erxyi.xxiang1;
            }
            else if (c == 's')
            {
                b = E_erxyi.xxiang2;
            }
            else if (c == 'j' && b == E_erxyi.xxiang1 && a == E_changjingenum.kaishichangjing)
            {
                a = E_changjingenum.youxichangjing;
            }
            else if (c == 'j' && b == E_erxyi.xxiang1 && a == E_changjingenum.jieshuchangjing)
            {
                a = E_changjingenum.kaishichangjing;
            }
            else if (c == 'j' && b == E_erxyi.xxiang2)
            {
                Environment.Exit(0);
            }
        }
    }

    /// <summary>

    class DiTu 
    {
        public int gezi;
        public HoQiang[] hqmen;
        public int dituwz =0;
        public void huizhi(int h,int w)//绘制地图方法
        {
            gezi = (2*h)+(w-2)*2;
            hqmen = new HoQiang[gezi];
            for (int i = 0; i < h; i++) //画竖着的两条地图
            {
                hqmen[dituwz] = new HoQiang();//我出错的地方，hqmen数组初始化了但是数组中的类对象也需要初始化
                hqmen[dituwz].wzxy.x = 0;
                hqmen[dituwz].wzxy.y = i;
                hqmen[dituwz].huizhi();
                dituwz++;
                hqmen[dituwz] = new HoQiang();
                hqmen[dituwz].wzxy.x = w-1;
                hqmen[dituwz].wzxy.y = i;
                hqmen[dituwz].huizhi();
                dituwz++;
            }
            for (int i = 1; i < w-1; i++) //画横着的两条地图
            {
                hqmen[dituwz] = new HoQiang();
                hqmen[dituwz].wzxy.x = i;
                hqmen[dituwz].wzxy.y = 0;
                hqmen[dituwz].huizhi();
                dituwz++;
                hqmen[dituwz] = new HoQiang();
                hqmen[dituwz].wzxy.x = i;
                hqmen[dituwz].wzxy.y = h - 1;
                hqmen[dituwz].huizhi();
                dituwz++;
            }
            dituwz = 0;
        }
    }

    /// <summary>
    /// 贪吃蛇类
    /// </summary>
    class TanChSheLei :Izhengengxin
    {
        public SheLei[] sheLeis = new SheLei[1];
        public E_fangxiangenum fx = E_fangxiangenum.xia;
        public int yidocshu;
        public void panduanfangxiang() //控制贪吃蛇方向的方法
        {
            if (Console.KeyAvailable) //检测键盘输入 ，有输入进入if
            {
                char c =Console.ReadKey(true).KeyChar;

                switch (c)//改变方向状态
                {
                    case 'w':
                        if (fx == E_fangxiangenum.xia)
                        {
                            fx = E_fangxiangenum.xia;
                        }
                        else 
                        {
                            fx = E_fangxiangenum.shang;
                        }
                        break;
                    case 'a':
                        if (fx == E_fangxiangenum.you)
                        {
                            fx = E_fangxiangenum.you;
                        }
                        else
                        {
                            fx = E_fangxiangenum.zuo;
                        }
                        break;
                    case 's':
                        if (fx == E_fangxiangenum.shang)
                        {
                            fx = E_fangxiangenum.shang;
                        }
                        else
                        {
                            fx = E_fangxiangenum.xia;
                        }
                        break;
                    case 'd':

                        if (fx == E_fangxiangenum.zuo)
                        {
                            fx = E_fangxiangenum.zuo;
                        }
                        else
                        {
                            fx = E_fangxiangenum.you;
                        }
                        break;
                }
            }
        }
        public void yido() //实现移动的方法
        {
            for (int i = 0; i < sheLeis.Length; i++) //清除贪吃蛇最后一个位置
            {
                Console.SetCursorPosition(sheLeis[sheLeis.Length-1].wzxy.x, sheLeis[sheLeis.Length - 1].wzxy.y);
                Console.Write(" ");
            }
            for (int i = sheLeis.Length-1; i > 0; i--) //实现除了贪吃蛇头以外的身子移动
            {
                sheLeis[i].wzxy.x = sheLeis[i-1].wzxy.x;
                sheLeis[i].wzxy.y = sheLeis[i - 1].wzxy.y;
            }
            switch (fx) //实现贪吃蛇头的移动
            {
                case E_fangxiangenum.shang:
                    sheLeis[0].wzxy.y--;
                    break;
                case E_fangxiangenum.xia:
                    sheLeis[0].wzxy.y++;
                    break;
                case E_fangxiangenum.zuo:
                    sheLeis[0].wzxy.x--;
                    break;
                case E_fangxiangenum.you:
                    sheLeis[0].wzxy.x++;
                    break;
            }
        }
        public void tcschsw(ShiWuShuZuLei sw) //贪吃蛇吃到食物的方法
        {
            if (sw.swshuzu != null ) 
            {
                for (int i = 0; i < sw.swshuzu.Length; i++)
                {
                    if (sheLeis[0].wzxy.x == sw.swshuzu[i].wzxy.x && sheLeis[0].wzxy.y == sw.swshuzu[i].wzxy.y) //检测贪吃蛇是否吃到食物
                    {
                        SheLei[] a = sheLeis;
                        sheLeis = new SheLei[sheLeis.Length + 1];
                        for (int j = 0; j < a.Length; j++)
                        {
                            sheLeis[j] = a[j];
                        }
                        sheLeis[sheLeis.Length - 1] = new SheLei(); //当贪吃蛇吃到食物的时候增加一个蛇身子
                        sheLeis[sheLeis.Length - 1].wzxy.x = sheLeis[sheLeis.Length - 2].wzxy.x;
                        sheLeis[sheLeis.Length - 1].wzxy.y = sheLeis[sheLeis.Length - 2].wzxy.y;
                        switch (fx) //实现最后一个身子的位置
                        {
                            case E_fangxiangenum.shang:
                                sheLeis[sheLeis.Length - 1].wzxy.y++;
                                break;
                            case E_fangxiangenum.xia:
                                sheLeis[sheLeis.Length - 1].wzxy.y--;
                                break;
                            case E_fangxiangenum.zuo:
                                sheLeis[sheLeis.Length - 1].wzxy.x++;
                                break;
                            case E_fangxiangenum.you:
                                sheLeis[sheLeis.Length - 1].wzxy.x--;
                                break;
                        }

                        ShiWulei[] b = sw.swshuzu; //对食物数组减少
                        sw.swshuzu = new ShiWulei[sw.swshuzu.Length-1];
                        int z = 0;
                        for (int k = 0; k < sw.swshuzu.Length; k++)
                        {
                            if (k == i)
                            {
                                z++;
                            }
                            sw.swshuzu[k] = b[z];
                            z++;
                        }

                    }
                }
            }
            
        }
        public bool siwang() //贪吃蛇死亡的方法
        {
            if (sheLeis[0].wzxy.x==0 || sheLeis[0].wzxy.x == 119 || sheLeis[0].wzxy.y == 0 || sheLeis[0].wzxy.y == 29) //检测撞到地图死亡
            {
                return true;
            }
            for (int i = 1; i < sheLeis.Length-1; i++) 
            {
                if (sheLeis[0].wzxy.x == sheLeis[i].wzxy.x && sheLeis[0].wzxy.y == sheLeis[i].wzxy.y ) //检测撞到自己蛇身死亡
                {
                    return true;
                }
            }

            return false;
        }
        public void huizhi(ref int yidocs) //绘制贪吃蛇的方法
        {
            for (int i = 0; i < sheLeis.Length; i++)
            {
                sheLeis[i].huizhi();
            }
            yidocs++;
        }
        public void zhenkozhi(DateTime sj) //帧控制方法实现帧移动的关键
        {
            int sudu = 100;
            int sjc = (int)(DateTime.Now - sj).TotalMilliseconds;
            int kozhuansj = sudu - sjc;
            if (kozhuansj > 0) 
            {
                Thread.Sleep(kozhuansj);
            }
        }

    /// <summary>
    /// 食物数组类
    /// </summary>
    class ShiWuShuZuLei 
    {
        public ShiWulei[] swshuzu;
        public void ShengChengSW(TanChSheLei tcs) //随机生成食物方法
        {
            Random r = new Random();

            if (swshuzu == null) 
            {
                swshuzu = new ShiWulei[1];
                swshuzu[0] = new ShiWulei();
                swshuzu[0].wzxy.x = r.Next(1, 119);
                swshuzu[0].wzxy.y = r.Next(1, 29);
                swshuzu[0].huizhi();
            }

            ShiWulei[] roqi = swshuzu; //对食物数组增加
            swshuzu = new ShiWulei[swshuzu.Length + 1];
            swshuzu[swshuzu.Length-1] = new ShiWulei();

            for (int i = 0; i < roqi.Length; i++)
            {
                swshuzu[i] = roqi[i];
            }


            swshuzu[swshuzu.Length - 1].wzxy.x = r.Next(1,119); //随机生成位置x
            swshuzu[swshuzu.Length - 1].wzxy.y = r.Next(1, 29); //随机生成位置y
            for (int i = 0; i < tcs.sheLeis.Length; i++)
            {
                if (swshuzu[swshuzu.Length - 1].wzxy.x == tcs.sheLeis[i].wzxy.x && swshuzu[swshuzu.Length - 1].wzxy.y == tcs.sheLeis[i].wzxy.y) //当生成的位置有对象了重新随机
                {
                    swshuzu[swshuzu.Length - 1].wzxy.x = r.Next(1, 119);
                    swshuzu[swshuzu.Length - 1].wzxy.y = r.Next(1, 29);
                    i--;
                }
            }
            for (int i = 0; i < swshuzu.Length-1; i++)
            {
                if (swshuzu[swshuzu.Length - 1].wzxy.x == swshuzu[i].wzxy.x && swshuzu[swshuzu.Length - 1].wzxy.y == swshuzu[i].wzxy.y) //当生成的位置有对象了重新随机
                {
                    swshuzu[swshuzu.Length - 1].wzxy.x = r.Next(1, 119);
                    swshuzu[swshuzu.Length - 1].wzxy.y = r.Next(1, 29);
                    i--;
                }
            }
            swshuzu[swshuzu.Length - 1].huizhi(); //绘制生成的食物
        }
    }
