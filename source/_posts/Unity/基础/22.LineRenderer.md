### 知识点一 LineRenderer是什么
LineRenderer是Unity提供的一个用于画线的组件
使用它我们可以在场景中绘制线段
一般可以用于
1绘制攻击范围
2武器红外线
3辅助功能
4其它画线功能

### 知识点三 LineRender代码相关

``` CSharp
//动态添加一个线段
GameObject line = new GameObject();
line.name = "Line";
LineRenderer lineRenderer = line.AddComponent<LineRenderer>();

//首尾相连
lineRenderer.loop = true;

//开始结束宽
lineRenderer.startWidth = 0.02f;
lineRenderer.endWidth = 0.02f;

//开始结束颜色
lineRenderer.startColor = Color.white;
lineRenderer.endColor = Color.red;

//设置材质
m = Resources.Load<Material>("M");
lineRenderer.material = m;

//设置点
//一定注意 设置点 要 先设置点的个数
lineRenderer.positionCount = 4;
//接着就设置 对应每个点的位置
lineRenderer.SetPositions(new Vector3[] { new Vector3(0,0,0),
                                          new Vector3(0,0,5),
                                          new Vector3(5,0,5)});
lineRenderer.SetPosition(3, new Vector3(5, 0, 0));

//是否使用世界坐标系
//决定了 是否随对象移动而移动
lineRenderer.useWorldSpace = false;

//让线段受光影响 会接受光数据 进行着色器计算
lineRenderer.generateLightingData = true;
```

``` CSharp
private LineRenderer lr;
public int dianshu = 360;
private float jiaodu;
private Vector3 wzhi;
public float r = 2;
void Start()
{
    lr = this.AddComponent<LineRenderer>();
    lr.material = Resources.Load<Material>("textMaterial");
    lr.positionCount = dianshu;
    jiaodu = 360 / dianshu;
    lr.startWidth = 0.2f;
    lr.loop = true;
    lr.useWorldSpace = false;

    huayuan(Vector3.zero, r);
}
public void huayuan(Vector3 v, float r) 
{
    
    for (int i = 0; i < dianshu; i++) 
    {
        if (wzhi == Vector3.zero)
        {
            wzhi = new Vector3(v.x, v.y, v.z + r);
        }
        else 
        {
            wzhi = Quaternion.AngleAxis(jiaodu , Vector3.up) * wzhi;
        }
        lr.SetPosition(i, wzhi);
    }
}
```

``` CSharp
private LineRenderer lr;
private Vector3 shishidjwz;
void Start()
{
    lr = gameObject.AddComponent<LineRenderer>();
    lr.material = Resources.Load<Material>("textMaterial");
    lr.startWidth = 0.2f;
    lr.positionCount = 0;
}
void Update()
{
    if (Input.GetMouseButton(0)) 
    {
        shishidjwz = Input.mousePosition;
        shishidjwz.z = 5;
        lr.positionCount += 1;
        lr.SetPosition(lr.positionCount-1, Camera.main.ScreenToWorldPoint(shishidjwz));
    }
}
```
