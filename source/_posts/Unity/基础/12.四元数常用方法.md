public Transform a;
public Transform b;
public Transform c;
private float time;
private Quaternion q;

private void LateUpdate()
{
    b.position = Vector3.Lerp(b.position, a.position + a.forward * -4 + a.up * 7, Time.deltaTime);


    //b.rotation = Quaternion.LookRotation(a.position - b.position);

    //b.rotation = Quaternion.Slerp(b.rotation, a.rotation, Time.deltaTime);

    if (q != Quaternion.LookRotation(a.position - b.position)) 
    {
        q = Quaternion.LookRotation(a.position - b.position);
        time = 0;
    }
    time += Time.deltaTime;
    b.rotation = Quaternion.Slerp(b.rotation ,q , time);
}
