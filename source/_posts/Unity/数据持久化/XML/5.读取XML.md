#### 引用命名空间 using System.Xml
#### 将xml文件放置StreamingAssets文件夹中 (需要手动创建)

``` xml
<?xml version="1.0" encoding="utf-8"?>
<Player>
    <name>邓佳乐</name>
    <age>20</age>
    <Equips>
        <Equip id="101" count="3"/>
	</Equips>
    <Friend>
		<name>小明</name>
        <age>21</age>
	</Friend>
    <Friend>
		<name>小红</name>
        <age>19</age>
	</Friend>
</Player>
```

``` CSharp
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using System.Xml;

public class XmlDoctext : MonoBehaviour
{
    void Start()
    {
        //StreamingAssets下xml的路径
        string str = Application.streamingAssetsPath + "/xmltext.xml";
        //XmlDocument xml文件类 整个xml文件
        XmlDocument xmldoc = new XmlDocument();
        //加载xml方法 传文件路径
        xmldoc.Load(str);


        //读取根节点
        XmlNode root = xmldoc.SelectSingleNode("Player");
        //读取根节点下的子节点
        XmlNode name = root.SelectSingleNode("name");
        //读取节点文本内容
        string djl = name.InnerText;
        int age =int.Parse(root.SelectSingleNode("age").InnerText); //获得string转int


        //读取节点属性
        print(root.SelectSingleNode("Equips/Equip").Attributes["id"].Value);
        XmlNode equip = root.SelectSingleNode("Equips/Equip");
        int count = int.Parse(equip.Attributes["count"].Value);
        print(count);


        //读取相同名节点
        XmlNodeList friends = root.SelectNodes("Friend");
        //遍历同名节点  两种方法
        foreach (XmlNode friend in friends) 
        {
            print(friend.SelectSingleNode("name").InnerText);
            print(friend.SelectSingleNode("age").InnerText);
        }
        for (int i = 0; i < friends.Count; i++) 
        {
            print(friends[i].SelectSingleNode("name").InnerText);
            print(friends[i].SelectSingleNode("age").InnerText);
        }

    }
```

---
#### 总结
XmlDocument装载xml文件的类  XmlNode节点类  XmlNodeList节点数组类用来装同名节点的

XmlDocument.Load() 加载xml文件

SelectSingleNode("路径")  找节点  用XmlNode类装
InnerText  读取节点内容 返回string类型 
Attributes["属性名"].Value  读取属性 返回string类型
SelectNodes("路径")  读取同名节点 用XmlNodeList类装
