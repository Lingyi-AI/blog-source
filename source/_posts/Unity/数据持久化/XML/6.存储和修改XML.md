#### 引用命名空间 using System.IO;
System.IO 是 .NET 内置文件、文件夹、数据流操作库，不属于 XML 专用。
不管你用 XmlDocument / XDocument / 文本 / Json/XML，只要读写本地硬盘文件，几乎都要用到它。

#### XmlElement 添加节点属性类
#### XmlDeclaration 添加头部声明类

``` CSharp
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using System.Xml;
using System.IO;

public class xmldocduxie : MonoBehaviour
{
    void Start()
    {
        //创建xml
        XmlDocument xmldoc = new XmlDocument();
        //添加头部声明
        XmlDeclaration xmldec = xmldoc.CreateXmlDeclaration("1.0","UTF-8","");
        xmldoc.AppendChild(xmldec);
        //添加根节点
        XmlElement root = xmldoc.CreateElement("root");
        xmldoc.AppendChild(root);
        //为根节点添加子节点
        XmlElement name = xmldoc.CreateElement("name");
        name.InnerText = "邓佳乐";
        root.AppendChild(name);

        XmlElement age = xmldoc.CreateElement("age");
        age.InnerText = "20";
        root.AppendChild(age);
        //在根节点添加节点的其他方式
        XmlElement friends = xmldoc.CreateElement("friends");
        for (int i = 1; i < 4; i++) 
        {
            XmlElement friend = xmldoc.CreateElement("friend");
            friend.InnerText = "朋友" + i;
            friends.AppendChild(friend);
        }
        root.AppendChild(friends);
        //为节点添加属性
        XmlElement equips = xmldoc.CreateElement("equips");
        for (int i = 1; i < 4; i++) 
        {
            XmlElement equip = xmldoc.CreateElement("equip");
            equip.SetAttribute("id",i.ToString());
            equip.SetAttribute("atk",(i * 10).ToString());
            equips.AppendChild(equip);
        }
        root.AppendChild(equips);
        //保存  save方法是将数据写入硬盘  文件在方法中的路径中
        string path = Application.persistentDataPath + "/newxml.xml";
        xmldoc.Save(path);


        //判断文件路径是否存在方法  File.Exists("路径")
        if (File.Exists(path)) 
        {
            XmlDocument xmldoc2 = new XmlDocument();
            xmldoc2.Load(path);
            //修改节点内容
            xmldoc2.SelectSingleNode("root/age").InnerText = "21";
            //修改节点属性
            XmlNodeList xmllist = xmldoc2.SelectNodes("root/equips/equip");
            xmllist[2].Attributes["atk"].Value = "50";
            //添加新节点
            XmlElement fri = xmldoc2.CreateElement("friend");
            fri.InnerText = "朋友4";
            xmldoc2.SelectSingleNode("root/friends").AppendChild(fri);
            //移除节点  注意: 父节点.RemoveChild(子节点) 这条 API 有强制规则
            xmldoc2.SelectSingleNode("root/friends").RemoveChild(xmldoc2.SelectSingleNode("root/friends/friend"));
            //保存覆盖修改内容
            xmldoc2.Save(path);

        }

    }
}
```

### 总结
Save("路径") 保存文件方法 不调用就不会把数据写入硬盘
CreateElement("节点名") 创建元素节点
AppendChild("节点") 把节点挂载到文档 / 父节点上（新建节点必须 Append，否则不在树里）
SetAttribute("属性名","值")（XmlElement 专属快捷方法）  创建属性
父节点.RemoveChild(子节点) 删除节点
判断文件路径是否存在方法  File.Exists("路径")
